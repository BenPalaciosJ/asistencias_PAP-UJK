function main(workbook: ExcelScript.Workbook, correo: string, fechaHoraStr: string, pinIngresado: string, responseId: string) { // Validar que la fecha no venga vacía if (!fechaHoraStr || typeof fechaHoraStr !== 'string' || fechaHoraStr.trim() === "") { throw new Error("La fecha y hora recibidas están vacías o no son válidas."); } // Normalizar la cadena de fecha (formato ISO) let fechaLimpiaStr = fechaHoraStr.trim().replace(" ", "T"); let fechaEnvio = new Date(fechaLimpiaStr); if (isNaN(fechaEnvio.getTime())) { fechaEnvio = new Date(); } // Ajustar automáticamente de UTC a Hora de México (Restar 6 horas) fechaEnvio.setHours(fechaEnvio.getHours() - 6); let anio = fechaEnvio.getFullYear(); let mes = String(fechaEnvio.getMonth() + 1).padStart(2, '0'); let dia = String(fechaEnvio.getDate()).padStart(2, '0'); let horas = String(fechaEnvio.getHours()).padStart(2, '0'); let minutos = String(fechaEnvio.getMinutes()).padStart(2, '0'); let fechaLimpia = `${anio}-${mes}-${dia}`; // "YYYY-MM-DD" let horaLimpiaMinutos = parseInt(horas) * 60 + parseInt(minutos); let horaParaMostrar = `${horas}:${minutos}`; // Hora local corregida (Ej. 19:15) let jsDay = fechaEnvio.getDay(); let diaSemana = jsDay === 0 ? 7 : jsDay; // 1=Lunes ... 3=Miércoles, 4=Jueves // 1. Leer la tabla de Alumnos (Base_Alumnos / T_Alumnos) let sheetAlumnos = workbook.getWorksheet("Base_Alumnos"); let tablaAlumnos = sheetAlumnos.getTable("T_Alumnos"); let alumnosData = tablaAlumnos.getRange().getTexts(); let headersAlumnos = tablaAlumnos.getHeaderRowRange().getTexts()[0]; let colCorreo = headersAlumnos.indexOf("Correo_Institucional"); let colMateria = headersAlumnos.indexOf("Materia"); let colDia = headersAlumnos.indexOf("Dia"); let colInicio = headersAlumnos.indexOf("Hora_Inicio"); let colFin = headersAlumnos.indexOf("Hora_Fin"); let colNombre = headersAlumnos.indexOf("Nombre"); let nombreAlumno = "Desconocido"; let materiaEncontrada = ""; let matchEncontrado = false; function timeToMinutes(timeStr: string): number { if (!timeStr) return 0; let parts = timeStr.trim().substring(0, 5).split(":"); if (parts.length < 2) return 0; return parseInt(parts[0]) * 60 + parseInt(parts[1]); } // 2. Buscar alumno y materia por horario y día for (let i = 1; i < alumnosData.length; i++) { let row = alumnosData[i]; if (row[colCorreo].trim().toLowerCase() === correo.trim().toLowerCase()) { nombreAlumno = row[colNombre]; let diaClase = parseInt(row[colDia]); let horaInicioMin = timeToMinutes(row[colInicio]); let horaFinMin = timeToMinutes(row[colFin]); // Validar si coincide el día y si la hora de envío está en el rango de la clase if (diaClase === diaSemana && horaLimpiaMinutos >= horaInicioMin && horaLimpiaMinutos <= horaFinMin) { materiaEncontrada = row[colMateria]; matchEncontrado = true; break; } } } let resultadoFinal = "Falta"; let motivoValidacion = ""; let pinOficialStr = "-"; if (!matchEncontrado) { motivoValidacion = "Alumno fuera de horario o no encontrado"; } else { // 3. Buscar PIN oficial en Pines_Sesiones basándose únicamente en Fecha y Materia let sheetPines = workbook.getWorksheet("PINs_Sesiones"); let tablaPines = sheetPines.getTable("Pines_Sesiones"); let pinesData = tablaPines.getRange().getTexts(); let headersPines = tablaPines.getHeaderRowRange().getTexts()[0]; let colPFecha = headersPines.indexOf("Fecha_Sesion"); let colPMateria = headersPines.indexOf("Materia"); let colPPin = headersPines.indexOf("PIN"); let pinEncontrado = false; for (let j = 1; j < pinesData.length; j++) { let pRow = pinesData[j]; let pFecha = pRow[colPFecha].split('T')[0]; let pMateria = pRow[colPMateria].trim(); if (pFecha === fechaLimpia && pMateria.toLowerCase() === materiaEncontrada.toLowerCase()) { pinOficialStr = pRow[colPPin].toString().trim(); pinEncontrado = true; break; } } if (!pinEncontrado) { motivoValidacion = "PIN no programado para esta sesión"; } else if (pinIngresado.trim() === pinOficialStr) { resultadoFinal = "Asistencia Válida"; motivoValidacion = "Ninguno"; } else { motivoValidacion = "PIN Incorrecto"; } } // 4. Guardar el resultado en la tabla de auditoría (T_Respuestas / Respuestas_Crudas) let sheetRespuestas = workbook.getWorksheet("Respuestas_Crudas"); let tablaRespuestas = sheetRespuestas.getTable("T_Respuestas"); tablaRespuestas.addRow(-1, [ responseId, correo, `${fechaLimpia} ${horaParaMostrar}`, pinIngresado, nombreAlumno, materiaEncontrada, pinOficialStr, resultadoFinal, motivoValidacion ]); }
